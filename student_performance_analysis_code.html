all_student_data <- tibble(
  grade_point_dec = round(runif(200, 0, 4), 2),
  absence_rate = sample(c("low", "medium", "high", "very high"), 200, replace = TRUE),
  absences = sample(0:15, 200, replace = TRUE),       # Now 0–15
  tardies = sample(0:15, 200, replace = TRUE),         # Now 0–15
  ss_absences = sample(0:15, 200, replace = TRUE),     # Now 0–15
  ell = sample(0:1, 200, replace = TRUE),
  sped = sample(0:1, 200, replace = TRUE),
  sec_504 = sample(0:1, 200, replace = TRUE),
  tag = sample(0:1, 200, replace = TRUE),
  pass_or_fail = sample(c(0, 1), 200, replace = TRUE)
)

all_with_core <- all_student_data %>%

grade_by_absence <- all_student_data %>%
  group_by(absence_rate) %>%
  summarise(
    mean_grade_point = mean(grade_point_dec, na.rm = TRUE),

  mutate(

Finding 3: Support needs explains decrease in grades; TAG explains increase in grades

Graph 3: Grade Point vs. Absences by Support Status (Core Classes)

# Translate four binary support group variables into one categorical variable
core_data_with_support <- all_with_core %>%
  mutate(support_group = case_when(
    ell == 1 ~ "ELL",
    sped == 1 ~ "SPED",
    sec_504 == 1 ~ "504",
    tag == 1 ~ "TAG",
    TRUE ~ "None"
  ))

# Calculate slope of GPA ~ absences for each group
core_slope_labels <- core_data_with_support %>%
  group_by(support_group) %>%
  nest() %>%
  mutate(
    model = map(data, ~ lm(grade_point_dec ~ absences, data = .x)),
    tidied = map(model, tidy)
  ) %>%
  unnest(tidied) %>%
  filter(term == "absences") %>%
  mutate(slope = estimate) %>%
  { .[, c("support_group", "slope")] }

# Choose label position
core_label_positions <- core_data_with_support %>%
  group_by(support_group) %>%
  summarize(x = max(absences, na.rm = TRUE) * 0.7, y = min(grade_point_dec, na.rm = TRUE) + 0.3)

# Merge slope values with label positions
core_slope_labels <- left_join(core_slope_labels, core_label_positions, by = "support_group")

# Plot the data
ggplot(core_data_with_support, aes(x = absences, y = grade_point_dec, color = support_group)) +
  geom_jitter(alpha = 0.3, width = 0.3, height = 0.1) +
  geom_smooth(method = "lm", se = FALSE) +
  geom_text(
    data = core_slope_labels,
    aes(x = x, y = y, label = paste0("slope = ", round(slope, 3))),
    inherit.aes = FALSE,
    color = "grey30",
    size = 3,
    vjust = -0.5
  ) +
  facet_wrap(~ support_group) +
  scale_y_continuous() +  # optional
  labs(
    title = "Grade Point vs. Absences by Support Status (Core Classes)",
    x = "Absences for Class",
    y = "Grade Point for Class"
  ) +
  theme_minimal() +
  theme(legend.position = "none")

    )
  )

non_core_slope_labels <- non_core_data_with_support %>%
  group_by(support_group) %>%
  nest() %>%
  mutate(
    model = map(data, ~ lm(grade_point_dec ~ absences, data = .x)),
    tidied = map(model, tidy)
  ) %>%
  unnest(tidied) %>%
  filter(term == "absences") %>%
  mutate(slope = estimate) %>%
  { .[, c("support_group", "slope")] }

# Choose label position
non_core_label_positions <- non_core_data_with_support %>%
  group_by(support_group) %>%
  summarize(x = max(absences, na.rm = TRUE) * 0.7, y = min(grade_point_dec, na.rm = TRUE) + 0.3)

# Merge slope values with label positions
non_core_slope_labels <- left_join(non_core_slope_labels, non_core_label_positions, by = "support_group")

# Plot the data
ggplot(non_core_data_with_support, aes(x = absences, y = grade_point_dec, color = support_group)) +
  geom_jitter(alpha = 0.3, width = 0.3, height = 0.1) +
  geom_smooth(method = "lm", se = FALSE) +
  geom_text(
    data = non_core_slope_labels,
    aes(x = x, y = y, label = paste0("slope = ", round(slope, 3))),
    inherit.aes = FALSE,
    color = "grey30",
    size = 3,
    vjust = -0.5
  ) +
  facet_wrap(~ support_group) +
  scale_y_continuous() +  # optional
  labs(
    title = "Grade Point vs. Absences by Support Status (Non-core Classes)",
    x = "Absences for Class",
    y = "Grade Point for Class"
  ) +
  theme_minimal() +
  theme(legend.position = "none")
